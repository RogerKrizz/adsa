#include <stdio.h>
#include <limits.h>
#include <string.h> // Needed for strlen in Gantt chart

#define MAX_GANTT 1000 // A large number since processes can be interrupted many times

struct Process {
    int pid, at, bt, rt;
    int ct, tat, wt;
};

int main() {
    int n;
    printf("Enter number of processes: ");
    scanf("%d", &n);

    struct Process p[n];

    // Input and setup
    for (int i = 0; i < n; i++) {
        p[i].pid = i + 1;
        printf("Enter AT and BT for P%d: ", i + 1);
        scanf("%d %d", &p[i].at, &p[i].bt);
        p[i].rt = p[i].bt; // Remaining time starts equal to burst time
    }

    int time = 0;
    int completed = 0;
    int min_rt;
    int shortest = -1;
    int is_idle;

    // Gantt Chart tracking variables
    int ganttPid[MAX_GANTT];
    int ganttStart[MAX_GANTT];
    int ganttEnd[MAX_GANTT];
    int ganttCount = 0;
    int prev_pid = -1; // Keeps track of who ran in the previous second
    int current_pid;

    // Main execution loop
    while (completed != n) {
        min_rt = INT_MAX;
        shortest = -1;
        is_idle = 1;

        // Find the process with the shortest remaining time that has arrived
        for (int i = 0; i < n; i++) {
            if (p[i].at <= time && p[i].rt > 0 && p[i].rt < min_rt) {
                min_rt = p[i].rt;
                shortest = i;
                is_idle = 0;
            }
        }

        // Determine who is getting the CPU right now (0 represents IDLE)
        current_pid = is_idle ? 0 : p[shortest].pid;

        // Context Switch Detected: If the process changed from the last second
        if (current_pid != prev_pid) {
            // If there was a previous process running, log its end time
            if (prev_pid != -1) {
                ganttEnd[ganttCount - 1] = time;
            }
            // Start a new segment in the Gantt chart
            ganttPid[ganttCount] = current_pid;
            ganttStart[ganttCount] = time;
            ganttCount++;
            
            // Update the previous process tracker
            prev_pid = current_pid; 
        }

        // Execute the CPU
        if (is_idle) {
            time++;
            continue;
        }

        // If not idle, execute the shortest process for 1 unit of time
        p[shortest].rt--;
        time++;

        // If the process just finished
        if (p[shortest].rt == 0) {
            completed++;
            p[shortest].ct = time;
            p[shortest].tat = p[shortest].ct - p[shortest].at;
            p[shortest].wt = p[shortest].tat - p[shortest].bt;
        }
    }

    // Close the very last Gantt chart segment when all processes are done
    if (ganttCount > 0) {
        ganttEnd[ganttCount - 1] = time;
    }

    // Process table and Averages
    float total_tat = 0, total_wt = 0;
    
    printf("\nPID\tAT\tBT\tCT\tTAT\tWT\n");
    for (int i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               p[i].pid, p[i].at, p[i].bt,
               p[i].ct, p[i].tat, p[i].wt);
               
        total_tat += p[i].tat;
        total_wt += p[i].wt;
    }

    printf("\nAverage Turnaround Time: %.2f", total_tat / n);
    printf("\nAverage Waiting Time: %.2f\n", total_wt / n);

    // ==========================================
    // DYNAMIC PROPORTIONAL GANTT CHART
    // ==========================================
    printf("\n\nGANTT CHART\n\n");

    int widths[MAX_GANTT];

    // 1. Calculate proportional widths based on duration
    for (int i = 0; i < ganttCount; i++) {
        int duration = ganttEnd[i] - ganttStart[i];
        widths[i] = duration * 2; // Scale factor: 1 unit of time = 2 spaces
        if (widths[i] < 6) {
            widths[i] = 6; // Minimum width so text like "IDLE" doesn't break UI
        }
    }

    // 2. Print Top Border
    printf(" ");
    for (int i = 0; i < ganttCount; i++) {
        for (int j = 0; j < widths[i]; j++) printf("-");
        printf(" ");
    }
    printf("\n");

    // 3. Print Middle (Process IDs perfectly centered)
    printf("|");
    for (int i = 0; i < ganttCount; i++) {
        char label[10];
        if (ganttPid[i] == 0) sprintf(label, "IDLE");
        else sprintf(label, "P%d", ganttPid[i]);
        
        int pad_total = widths[i] - strlen(label);
        int pad_left = pad_total / 2;
        int pad_right = pad_total - pad_left;
        
        for(int j = 0; j < pad_left; j++) printf(" ");
        printf("%s", label);
        for(int j = 0; j < pad_right; j++) printf(" ");
        printf("|");
    }
    printf("\n");

    // 4. Print Bottom Border
    printf(" ");
    for (int i = 0; i < ganttCount; i++) {
        for (int j = 0; j < widths[i]; j++) printf("-");
        printf(" ");
    }
    printf("\n");

    // 5. Print Timestamps perfectly aligned under the pipes '|'
    int chars_printed = printf("%d", ganttStart[0]);
    int target_pos = 0;
    
    for (int i = 0; i < ganttCount; i++) {
        target_pos += widths[i] + 1; // +1 to account for the pipe '|'
        
        // Print spaces until we reach exactly under the next pipe
        while (chars_printed < target_pos) {
            printf(" ");
            chars_printed++;
        }
        
        // Print the next timestamp and add its length to chars_printed
        int num_len = printf("%d", ganttEnd[i]);
        chars_printed += num_len;
    }
    printf("\n\n");

    return 0;
}
