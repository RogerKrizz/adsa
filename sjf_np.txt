#include <stdio.h>
#include <limits.h>
#include <string.h>

#define MAX_GANTT 100 // A smaller max is fine since there are no interruptions

struct Process {
    int pid, at, bt;
    int ct, tat, wt;
    int is_completed; // 0 means not done, 1 means done
};

int main() {
    int n;
    printf("Enter number of processes: ");
    scanf("%d", &n);

    struct Process p[n];

    // Input and setup
    for (int i = 0; i < n; i++) {
        p[i].pid = i + 1;
        printf("Enter AT and BT for P%d: ", i + 1);
        scanf("%d %d", &p[i].at, &p[i].bt);
        p[i].is_completed = 0; // Initially, no process is completed
    }

    int time = 0;
    int completed = 0;
    
    // Gantt chart arrays
    int ganttPid[MAX_GANTT];
    int ganttStart[MAX_GANTT];
    int ganttEnd[MAX_GANTT];
    int ganttCount = 0;

    // Main execution loop
    while (completed < n) {
        int min_bt = INT_MAX;
        int shortest = -1;

        // Find the process with the shortest Burst Time that has arrived
        for (int i = 0; i < n; i++) {
            if (p[i].at <= time && p[i].is_completed == 0) {
                // If we find a shorter burst time, OR if it's a tie, pick the one that arrived first
                if (p[i].bt < min_bt || (p[i].bt == min_bt && p[i].at < p[shortest].at)) {
                    min_bt = p[i].bt;
                    shortest = i;
                }
            }
        }

        // If no process has arrived by the current time, the CPU is IDLE
        if (shortest == -1) {
            // Find the exact time the next process will arrive so we can fast-forward to it
            int next_arrival = INT_MAX;
            for (int i = 0; i < n; i++) {
                if (p[i].is_completed == 0 && p[i].at < next_arrival) {
                    next_arrival = p[i].at;
                }
            }

            // Log the IDLE time in the Gantt chart
            ganttPid[ganttCount] = 0; // 0 represents IDLE
            ganttStart[ganttCount] = time;
            ganttEnd[ganttCount] = next_arrival;
            ganttCount++;

            // Fast-forward the CPU clock to the next arrival time
            time = next_arrival; 
        } 
        else {
            // Log the process execution in the Gantt chart
            ganttPid[ganttCount] = p[shortest].pid;
            ganttStart[ganttCount] = time;
            
            // Execute the process to completion (Non-Preemptive)
            time += p[shortest].bt;
            
            ganttEnd[ganttCount] = time;
            ganttCount++;

            // Calculate metrics and mark as completed
            p[shortest].ct = time;
            p[shortest].tat = p[shortest].ct - p[shortest].at;
            p[shortest].wt = p[shortest].tat - p[shortest].bt;
            p[shortest].is_completed = 1;
            
            completed++;
        }
    }

    // Process table and Averages calculation
    float total_tat = 0, total_wt = 0;
    
    printf("\nPID\tAT\tBT\tCT\tTAT\tWT\n");
    for (int i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               p[i].pid, p[i].at, p[i].bt,
               p[i].ct, p[i].tat, p[i].wt);
               
        total_tat += p[i].tat;
        total_wt += p[i].wt;
    }

    printf("\nAverage Turnaround Time: %.2f", total_tat / n);
    printf("\nAverage Waiting Time: %.2f\n", total_wt / n);

    // ==========================================
    // DYNAMIC PROPORTIONAL GANTT CHART
    // ==========================================
    printf("\n\nGANTT CHART\n\n");

    int widths[MAX_GANTT];

    // 1. Calculate proportional widths based on duration
    for (int i = 0; i < ganttCount; i++) {
        int duration = ganttEnd[i] - ganttStart[i];
        widths[i] = duration * 2; // Scale factor: 1 unit of time = 2 spaces
        if (widths[i] < 6) {
            widths[i] = 6; // Minimum width so text like "IDLE" doesn't break the UI
        }
    }

    // 2. Print Top Border
    printf(" ");
    for (int i = 0; i < ganttCount; i++) {
        for (int j = 0; j < widths[i]; j++) printf("-");
        printf(" ");
    }
    printf("\n");

    // 3. Print Middle (Process IDs perfectly centered)
    printf("|");
    for (int i = 0; i < ganttCount; i++) {
        char label[10];
        if (ganttPid[i] == 0) sprintf(label, "IDLE");
        else sprintf(label, "P%d", ganttPid[i]);
        
        int pad_total = widths[i] - strlen(label);
        int pad_left = pad_total / 2;
        int pad_right = pad_total - pad_left;
        
        for(int j = 0; j < pad_left; j++) printf(" ");
        printf("%s", label);
        for(int j = 0; j < pad_right; j++) printf(" ");
        printf("|");
    }
    printf("\n");

    // 4. Print Bottom Border
    printf(" ");
    for (int i = 0; i < ganttCount; i++) {
        for (int j = 0; j < widths[i]; j++) printf("-");
        printf(" ");
    }
    printf("\n");

    // 5. Print Timestamps perfectly aligned under the pipes '|'
    int chars_printed = printf("%d", ganttStart[0]);
    int target_pos = 0;
    
    for (int i = 0; i < ganttCount; i++) {
        target_pos += widths[i] + 1; // +1 to account for the pipe '|'
        
        // Print spaces until we reach exactly under the next pipe
        while (chars_printed < target_pos) {
            printf(" ");
            chars_printed++;
        }
        
        // Print the next timestamp and add its length to chars_printed
        int num_len = printf("%d", ganttEnd[i]);
        chars_printed += num_len;
    }
    printf("\n\n");

    return 0;
}
